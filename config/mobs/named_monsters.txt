     # Named Monsters Configuration
     # Format:
     # [MONSTER]
     # name=Monster Name
     # encounter_day=100
     # base_health=200
     # base_attack=25
     # base_defense=15
     # base_speed=12
     # strategy=TACTICAL
     # equipment_type=LAPTOP
     # equipment_rarity=EPIC
     # [PATENT]
     # name=Patent Name
     # type=ATTACK/HEAL/DEFENSE/BUFF/DEBUFF
     # power=50
     # cost=15
     # target=SELF/ENEMY
     # hustle_mod=10
     # technical_debt_mod=-5
     # duration=3
     # [END]
